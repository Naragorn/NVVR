cmake_minimum_required(VERSION 3.20)
project(NVVR VERSION 0.0.14 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "NVVR requires Windows x86; configure with -A Win32")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

target_include_directories(NVVR PRIVATE src)
target_compile_options(NVVR PRIVATE /W4 /WX /permissive-)
target_link_libraries(NVVR PRIVATE obvr013 user32 nvvr_render)
enable_testing()
add_executable(nvvr_tests tests/Tests.cpp)
target_include_directories(nvvr_tests PRIVATE src)
target_compile_options(nvvr_tests PRIVATE /W4 /WX /permissive-)
